// Scratch (not committed): 50 varied playthroughs, writing the ordered choices.
import 'dart:convert';
import 'dart:io';
import 'package:flutter_test/flutter_test.dart';
import 'package:narrative_data_app/data/story_repository.dart';
import 'package:narrative_data_app/models/story_node.dart';
import 'package:narrative_data_app/screens/playthrough_simulator_screen.dart';

Map<String, dynamic> _table(String n) =>
    jsonDecode(File('assets/gamedata/$n.json').readAsStringSync()) as Map<String, dynamic>;

void main() {
  test('audit runs', () {
    final raw = jsonDecode(File(StoryRepository.assetPath).readAsStringSync()) as Map<String, dynamic>;
    final story = StoryData({for (final e in raw.entries) e.key: StoryNode.fromJson(e.key, e.value as Map<String, dynamic>)});
    final tables = {for (final n in ['enemies','dice','skills','items','shops','races','professions','spells','item_sets','zones','skill_trees','factions','signs','subclans','relations','titles','geography','biomes','quests','npcs','land_routes','land_events','houses','expedition_steps']) n: _table(n)};
    final professions = tables['professions']!.keys.toList()..sort();
    final races = tables['races']!.keys.toList()..sort();
    final strategies = SimStrategy.values;
    final plan = [
      for (var i = 0; i < 50; i++)
        (strategy: strategies[i % strategies.length], raceId: races[(i + i ~/ 5) % races.length] as String?, professionId: professions[i % professions.length] as String?),
    ];
    final seed = 20261015;
    final res = simulatePlaythroughs(story, tables: tables, gameConfig: _table('game_config'), plan: plan, seed: seed);
    File(Platform.environment['SIM_OUT']!).writeAsStringSync(jsonEncode([
      for (var i = 0; i < res.length; i++) {
        'run': i, 'seed': seed * 100003 + i, 'seedBase': seed,
        'strategy': res[i].strategy.name, 'race': res[i].raceId, 'profession': res[i].professionId,
        'level': res[i].finalLevel, 'gold': res[i].finalGold, 'alignment': res[i].finalAlignment,
        'endedHow': res[i].endedHow, 'chapter': res[i].furthestChapter, 'steps': res[i].steps,
        'choices': res[i].choicesTaken, 'flags': res[i].flagIds.toList(),
        'quests': res[i].questIds.toList(),
        'questRecords': [for (final q in res[i].questRecords) {'id': q.questId, 'from': q.acceptedAtStep, 'to': q.completedAtStep, 'turnIn': q.turnIn}],
        'questsStuck': res[i].questsStuck, 'shops': res[i].shopIds.toList(),
      }
    ]));
  }, timeout: const Timeout(Duration(minutes: 30)));
}
