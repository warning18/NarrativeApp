// Throw-away picture test for docs/maps/sphere-and-locations. Not part of
// the suite: it only runs with ATLAS=<folder>.
//   ATLAS=docs/maps/atlas/shots ATLAS_THEME=dark ATLAS_W=412 \
//     flutter test test/zz_atlas_shot_test.dart
// It opens the real app, jumps the story to a scene, and walks the map
// levels with the same buttons the player has, saving the screen each time.
import 'dart:io';
import 'dart:ui' as ui;

import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter/services.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:shared_preferences/shared_preferences.dart';

import 'package:narrative_data_app/data/map_charts.dart' show ChartPalette;
import 'package:narrative_data_app/data/room_plan.dart';
import 'package:narrative_data_app/data/world_map.dart';
import 'package:narrative_data_app/theme/stitched_ink.dart';
import 'package:narrative_data_app/widgets/room_plan_painter.dart';
import 'package:narrative_data_app/l10n/app_locale.dart';
import 'package:narrative_data_app/main.dart';
import 'package:narrative_data_app/providers/app_mode_provider.dart';
import 'package:narrative_data_app/providers/map_look_provider.dart';
import 'package:narrative_data_app/providers/player_session_provider.dart';
import 'package:narrative_data_app/providers/story_providers.dart';
import 'package:narrative_data_app/providers/theme_mode_provider.dart';
import 'package:narrative_data_app/screens/journey_screen.dart';
import 'package:narrative_data_app/providers/home_tab_provider.dart'
    show isGameRoute;

import 'support/fonts.dart';
import 'support/open_game.dart';
import 'support/pump.dart';

final _shotKey = GlobalKey();

void main() {
  final dir = Platform.environment['ATLAS'];
  if (dir == null) {
    test('the atlas pictures are taken only with ATLAS', () {});
    return;
  }
  final theme = Platform.environment['ATLAS_THEME'] ?? 'dark';
  final w = double.parse(Platform.environment['ATLAS_W'] ?? '412');
  final only = Platform.environment['ATLAS_ONLY'];
  final tag = '$theme-${w.round()}';
  Directory(dir).createSync(recursive: true);
  final log = File('$dir/log-$tag.txt');

  Future<void> shoot(WidgetTester tester, String name) async {
    await tester.runAsync(() async {
      final boundary =
          _shotKey.currentContext!.findRenderObject()! as RenderRepaintBoundary;
      final image = await boundary.toImage(pixelRatio: 2);
      final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
      File('$dir/$name-$tag.png').writeAsBytesSync(bytes!.buffer.asUint8List());
    });
  }

  Future<void> settle(WidgetTester tester, [int seconds = 2]) async {
    await settleApp(tester);
    await tester.pump(Duration(seconds: seconds));
    await tester.pump(const Duration(milliseconds: 400));
  }

  Future<bool> tap(WidgetTester tester, String key, {String? as}) async {
    final f = find.byKey(ValueKey(key));
    if (f.evaluate().isEmpty) {
      log.writeAsStringSync('MISSING $key (${as ?? ''})\n',
          mode: FileMode.append);
      return false;
    }
    await tester.tap(f.first, warnIfMissed: false);
    await settle(tester);
    return true;
  }

  Future<ProviderContainer> open(WidgetTester tester, String scene,
      {List<String> flags = const [], List<String> via = const []}) async {
    TestDefaultBinaryMessengerBinding.instance.defaultBinaryMessenger
        .setMockMethodCallHandler(
            const MethodChannel('flutter_tts'), (call) async => 1);
    SharedPreferences.setMockInitialValues({'tutorial_enabled': false});
    usePhone(tester, Size(w, 915));
    await tester.pumpWidget(ProviderScope(
      overrides: [storedCityPlans],
      child: RepaintBoundary(key: _shotKey, child: const MyApp()),
    ));
    await tester.runAsync(loadAppFonts);
    await settleApp(tester);
    await tester.tap(find.byKey(const Key('menu_edit_mode')));
    await settleApp(tester);
    final container =
        ProviderScope.containerOf(tester.element(find.byType(MaterialApp)));
    await tester.runAsync(() => container
        .read(appLanguageProvider.notifier)
        .setLanguage(AppLanguage.en));
    await tester.runAsync(() => container
        .read(themeModeProvider.notifier)
        .setThemeMode(theme == 'light' ? ThemeMode.light : ThemeMode.dark));
    await tester.runAsync(() => container
        .read(mapLookProvider.notifier)
        .choose(theme == 'light' ? MapLook.parchment : MapLook.night));
    await tester.runAsync(() => container
        .read(playerSessionProvider.notifier)
        .loadSession(PlayerSession.fromJson({
          'raceId': 'human',
          'professionId': 'warrior',
          'gold': 100,
          'flags': ['journey_first_shown', ...flags],
        })));
    await tester.runAsync(
        () => container.read(appModeProvider.notifier).setMode(AppMode.inGame));
    await settleApp(tester);
    await tester.tap(find.text('Journey'));
    await settleApp(tester);
    if (scene == 'mid') {
      // A scene of each landmark of chapters 1 to 3: most lands charted.
      final ids = [
        for (final l in worldMapLandmarks)
          if (l.chapter <= 3 && l.scenes.isNotEmpty) l.scenes.first,
      ];
      container
          .read(storyPlayProvider.notifier)
          .loadState(ids.last, ids.sublist(0, ids.length - 1));
      await settleApp(tester);
      await tester.pump(const Duration(seconds: 6));
    }
    for (final s in [...via, if (scene != 'mid') scene]) {
      container.read(storyPlayProvider.notifier).jumpTo(s);
      await settleApp(tester);
      await tester.pump(const Duration(seconds: 6));
      for (var i = 0; i < 3; i++) {
        Navigator.of(tester.element(find.byType(JourneyScreen)))
            .popUntil(isGameRoute);
        await tester.pump(const Duration(seconds: 1));
      }
    }
    await tester.pump(const Duration(milliseconds: 700));
    return container;
  }


  // The sphere as the only world map: how it opens, and closing in.
  for (final (name, scene, via) in [
    ('alster', '100', <String>[]),
    ('saltmouth', '2015_cards', ['2010']),
    ('mid', 'mid', <String>[]),
  ]) {
    if (only != null && !only.split(',').contains(name)) continue;
    testWidgets('sphere from $name', (tester) async {
      await open(tester, scene, via: via, flags: ['lysa_survived']);
      final pre = Platform.environment['ATLAS_TAG'] ?? 'before';
      await tap(tester, 'journey_tab_world');
      await shoot(tester, '$pre-$name-1-world');
      await tap(tester, 'journey_zoom_in');
      await shoot(tester, '$pre-$name-2-in1');
      await tap(tester, 'journey_zoom_in');
      await shoot(tester, '$pre-$name-3-in2');
      await tap(tester, 'journey_zoom_out');
      await tap(tester, 'journey_zoom_out');
      await tap(tester, 'journey_full_map');
      await shoot(tester, '$pre-$name-4-full');
      await tap(tester, 'journey_zoom_in');
      await shoot(tester, '$pre-$name-5-full-in1');
      tester.takeException();
    });
  }

  // From the sphere to the town and back, frame by frame, as it is today.
  if (only != null && only.split(',').contains('frames')) {
    testWidgets('frames', (tester) async {
      await open(tester, '2015_cards', via: ['2010']);
      await tap(tester, 'journey_tab_town');
      await shoot(tester, 'frames-0-town');
      // Up: the World tab.
      await tester.tap(find.byKey(const ValueKey('journey_tab_world')));
      for (var i = 1; i <= 6; i++) {
        await tester.pump(const Duration(milliseconds: 80));
        await shoot(tester, 'frames-up-$i');
      }
      await settle(tester);
      // Down: "+" to the closest zoom, then once more.
      for (var i = 0; i < 3; i++) {
        await tap(tester, 'journey_zoom_in');
      }
      await shoot(tester, 'frames-down-0');
      await tester.tap(find.byKey(const ValueKey('journey_zoom_in')));
      for (var i = 1; i <= 7; i++) {
        await tester.pump(const Duration(milliseconds: 80));
        await shoot(tester, 'frames-down-$i');
      }
      await settle(tester);
      await shoot(tester, 'frames-down-8');
      tester.takeException();
    });
  }

  if (only != null && only.split(',').contains('rungs')) {
    testWidgets('rungs', (tester) async {
      await open(tester, '100');
      Future<void> rung(String? r) async {
        debugAtlasRung = r;
        tester.element(find.byType(JourneyScreen)).markNeedsBuild();
        await tester.pump();
        await settle(tester);
      }

      await rung('room');
      await shoot(tester, 'rungs-1-room');
      await rung(null);
      await tap(tester, 'journey_place_zoom_out');
      await shoot(tester, 'rungs-2-district');
      await tap(tester, 'journey_tab_town');
      await shoot(tester, 'rungs-3-town');
      await tap(tester, 'journey_tab_world');
      await shoot(tester, 'rungs-4-world');
      tester.takeException();
    });
  }
}
