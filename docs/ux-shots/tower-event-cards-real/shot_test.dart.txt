// start screen (as it opens and with a pick), the marks and the end of a run
// (a death and a win), at 360 px in French and 412 px in English. Drawn only
// with UX_SHOTS.

Future<void> _loadFonts() async {
  for (final (family, files) in [
    (
      InkFonts.system,
      ['PixelifySans-Regular.ttf', 'PixelifySans-SemiBold.ttf']
    ),
    (
      InkFonts.prose,
      ['Spectral-Regular.ttf', 'Spectral-Medium.ttf', 'Spectral-Italic.ttf']
    ),
    (InkFonts.display, ['IMFellEnglishSC-Regular.ttf']),
  ]) {
    final loader = FontLoader(family);
    for (final file in files) {
      loader.addFont(Future.value(
          ByteData.sublistView(File('assets/fonts/$file').readAsBytesSync())));
    }
    await loader.load();
  }
  final icons = File('/root/fl/sdk/bin/cache/artifacts/material_fonts/'
      'MaterialIcons-Regular.otf');
  if (icons.existsSync()) {
    final loader = FontLoader('MaterialIcons')
      ..addFont(Future.value(ByteData.sublistView(icons.readAsBytesSync())));
    await loader.load();
  }
}

void main() {
  final shots = Platform.environment['UX_SHOTS'];
  testWidgets('the Tower screens', (tester) async {
    await tester.runAsync(_loadFonts);
    final key = GlobalKey();
    Future<void> shot(String name) async {
      final b =
          key.currentContext!.findRenderObject()! as RenderRepaintBoundary;
      late Uint8List bytes;
      await tester.runAsync(() async {
        final image = await b.toImage(pixelRatio: 2);
        bytes = (await image.toByteData(format: ui.ImageByteFormat.png))!
            .buffer
            .asUint8List();
      });
      File('$shots/$name.png').writeAsBytesSync(bytes);
    }

    Future<void> settle() async {
      for (var i = 0; i < 12; i++) {
        await tester.runAsync(
            () => Future<void>.delayed(const Duration(milliseconds: 20)));
        await tester.pump(const Duration(milliseconds: 100));
      }
    }

    const run = TowerRun(
      seed: 1,
      raceId: 'human',
      professionId: 'warrior',
      floor: 17,
      health: 0,
      status: TowerStatus.dead,
    );
    TowerRun? failedWon;
    for (final (lang, width, tag) in [
      (AppLanguage.fr, 360.0, 'fr360'),
      (AppLanguage.en, 412.0, 'en412'),
    ]) {
      Future<void> open(Widget screen,
          {Map<String, Object> prefs = const {}}) async {
        SharedPreferences.setMockInitialValues(
            {'app_language': lang.name, ...prefs});
        usePhone(tester, Size(width, 780));
        final container = ProviderContainer();
        addTearDown(container.dispose);
        await container.read(appLanguageProvider.notifier).setLanguage(lang);
        await tester.pumpWidget(UncontrolledProviderScope(
          container: container,
          child: RepaintBoundary(
            key: key,
            child: MaterialApp(
                key: UniqueKey(), // a screen pushed before is dropped
                theme: buildAppTheme(container
                    .read(appPaletteProvider)
                    .schemeFor(Brightness.light)),
                home: screen),
          ),
        ));
        await settle();
      }

      Map<String, Object> saved(TowerRun r) =>
          {'tower_run_v1': jsonEncode(r.toJson())};
      Future<void> settleLong() async {
        for (var i = 0; i < 40; i++) {
          await tester.runAsync(
              () => Future<void>.delayed(const Duration(milliseconds: 20)));
          await tester.pump(const Duration(milliseconds: 100));
        }
      }

      // A floor that gained a level opens the Level Up first: drawn once,
      // then closed so the cards behind it show.
      var levelShot = false;
      Future<void> pastLevelUp() async {
        final screen = find.byType(LevelUpScreen);
        if (screen.evaluate().isEmpty) return;
        if (!levelShot) {
          levelShot = true;
          await tester.pump(const Duration(milliseconds: 600));
          await shot('12-levelup-$tag');
        }
        Navigator.of(tester.element(screen)).pop();
        await settleLong();
      }

      // The spoils sheet of a won floor, drawn then taken: the boon or
      // oath comes after it.
      Future<void> spoils(String? name) async {
        final take = find.byKey(const Key('loot_sheet_take'));
        if (take.evaluate().isEmpty) return;
        await tester.pump(const Duration(milliseconds: 600));
        if (name != null) await shot(name);
        await tester.tap(take);
        await settleLong();
      }

      await open(const TowerFloorScreen(),
          prefs: saved(run.copyWith(
              status: TowerStatus.climbing, floor: 3, health: 30)));
      await settleLong();
      final data = await tester.runAsync(() => ProviderScope.containerOf(
              tester.element(find.byType(TowerFloorScreen)))
          .read(towerDataProvider.future));
      TowerRun base(int seed, int floor) => TowerRun(
          seed: seed,
          raceId: 'human',
          professionId: 'warrior',
          floor: floor,
          health: 30,
          events: 1);
      TowerRun? failed, rivalRun, skipped, passedRun;
      for (var seed = 1; seed < 400; seed++) {
        for (var f = 1; f < 29; f++) {
          final r = base(seed, f);
          final kind = towerEventKindOf(r);
          if (kind == null) continue;
          if (kind == TowerEventKind.test) {
            final won = towerAfterFight(r, data!, won: true, healthLeft: 30);
            final after = towerPick(won, data, null);
            if (after.lastEvent?.passed == false && failed == null) {
              failed = after;
              failedWon = won;
            }
          } else if (kind == TowerEventKind.rival) {
            rivalRun ??= r;
            final won =
                towerAfterFight(r, data!, won: true, healthLeft: 30, rounds: 2);
            skipped ??= towerPick(won, data, null);
          }
        }
      }
      Future<void> pastLevel() async {
        final lvl = find.byType(LevelUpScreen);
        if (lvl.evaluate().isNotEmpty) {
          Navigator.of(tester.element(lvl)).pop();
          await settleLong();
        }
      }

      // 1: the setback on the cards screen.
      await open(const TowerFloorScreen(), prefs: saved(failedWon!));
      await settleLong();
      await pastLevel();
      final take = find.byKey(const Key('loot_sheet_take'));
      if (take.evaluate().isNotEmpty) {
        await tester.pump(const Duration(milliseconds: 600));
        await tester.tap(take);
        await settleLong();
      }
      await shot('real-1-setback-$tag');
      // 2: the rival before the fight.
      await open(const TowerFloorScreen(), prefs: saved(rivalRun!));
      await settleLong();
      await shot('real-2-rival-$tag');
      // 3: the fall.
      await open(const TowerFloorScreen(), prefs: saved(failed!));
      await settleLong();
      await pastLevel();
      await shot('real-3-fell-$tag');
      // 4: the skip.
      await open(const TowerFloorScreen(), prefs: saved(skipped!));
      await settleLong();
      await pastLevel();
      await shot('real-4-skipped-$tag');
    }
  }, skip: shots == null);
}
