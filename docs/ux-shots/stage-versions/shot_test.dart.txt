// equal slots a side, a head and icons for each fighter, a tap on a die to
// pick it (its line to the target shows) and a second to keep it, a double
// tap on the kept row to put it back, the keep final after a new roll, and
// the pop-in of an effect icon.

bool _dark = Platform.environment['DARK'] == '1';
Future<void> _enter(WidgetTester tester,
    {List<SideLayer>? land,
    double w = 412,
    String id = 'slum_thug',
    List<String> more = const [],
    EncounterModifiers? mods,
    String? plate}) async {
  SharedPreferences.setMockInitialValues({
    // The dice thrown at 4x: the tests wait a quarter of the throw (the
    // default speed is proven by dice_speed_test).
    'combat_dice_speed': 'quad',
    if (_dark) 'app_theme_mode': 'dark',
    'gamedb_dice': jsonEncode({
      ...gamedata('dice.json'),
      'board_die': {
        'diceName': 'board_die',
        'numberOfFaces': 1,
        'faces': [
          {
            'faceName': 'Strike',
            'type': 'Attack',
            'value': 4,
            'element': 'None'
          },
        ],
      },
    }),
  });
  usePhone(tester, Size(w, 860));
  await tester.pumpWidget(ProviderScope(
      overrides: [fightRandomProvider.overrideWithValue(() => Random(7))],
      child: const MyApp()));
  await settle(tester);
  final container =
      ProviderScope.containerOf(tester.element(find.byType(MaterialApp)));
  await tester.runAsync(() => container
      .read(playerSessionProvider.notifier)
      .loadSession(PlayerSession.fromJson({
        'raceId': 'human',
        'professionId': 'warrior',
        'ownedDiceIds': ['board_die'],
        'equippedDiceId': 'board_die',
        'maxHealth': 400,
        'currentHealth': 400,
      })));
  final thug = {
    ...gamedata('enemies.json')[id] as Map<String, dynamic>,
    'maxHealth': 900,
    'skillMoves': <dynamic>[],
  };
  final screen = FightScreen(
    enemyId: id,
    landLayers: land,
    enemy: thug,
    additionalEnemyIds: more,
    additionalEnemies: {
      for (final m in more)
        m: {
          ...gamedata('enemies.json')[m] as Map<String, dynamic>,
          'maxHealth': 900,
          'skillMoves': <dynamic>[]
        }
    },
    modifiers: mods ??
        const EncounterModifiers(
            forcedCondition: BattlefieldCondition.highGround,
            forcedGoal: FightGoal.slay),
  );
  tester.state<NavigatorState>(find.byType(Navigator).first).push(
      MaterialPageRoute<bool>(
          builder: (_) => plate == null
              ? screen
              : FightPlateOverride(
                  asset: plate,
                  boardStage: FightStage.hall,
                  floor: 1,
                  child: screen)));
  await settle(tester);
  Future<void> until(bool Function() f, [int n = 150]) async {
    for (var i = 0; i < n && !f(); i++) {
      await tester.runAsync(
          () => Future<void>.delayed(const Duration(milliseconds: 60)));
      await tester.pump(const Duration(milliseconds: 100));
    }
  }

  await until(() => shows('Enter Battle'));
  for (var i = 0; i < 12; i++) {
    await tester
        .runAsync(() => Future<void>.delayed(const Duration(milliseconds: 80)));
    await tester.pump(const Duration(milliseconds: 100));
  }
}

void main() {
  final out = Platform.environment['UX_SHOTS'];
  final which = Platform.environment['CASE'] ?? 'salt';
  setUp(FightSpriteImages.reset);
  testWidgets('fight $which', skip: out == null, (tester) async {
    await loadAppFonts();
    final (stage, place) = switch (which) {
      'glass' => (FightStage.glass, 'greyhithe'),
      'ash' => (FightStage.ash, 'emberwick'),
      'terraces' => (FightStage.terraces, 'akagiri'),
      'waves' => (FightStage.waves, 'the_crossing'),
      'hall' => (FightStage.hall, 'drowned_cloister_stair'),
      'harbour' => (FightStage.cityRoad, 'white_anchorage_quay'),
      'ashshared' => (FightStage.ash, null),
      _ => (FightStage.glass, null),
    };
    await _enter(tester,
        id: 'slum_thug',
        more: const ['den_bouncer'],
        mods: EncounterModifiers(
            placeId: place, forcedStage: stage, forcedGoal: FightGoal.slay));
    Future<void> shot(String name) async {
      final b = tester.renderObject<RenderRepaintBoundary>(
          find.byType(RepaintBoundary).first);
      await tester.runAsync(() async {
        final img = await b.toImage(pixelRatio: 1);
        final d = await img.toByteData(format: ui.ImageByteFormat.png);
        File('$out/$name.png').writeAsBytesSync(d!.buffer.asUint8List());
      });
    }

    final tag = '${_dark ? 'dark' : 'light'}';
    await shot('$which-$tag-setup');
    await tester.ensureVisible(find.text('Enter Battle'));
    await tester.pump();
    await tester.tap(find.text('Enter Battle'));
    for (var i = 0; i < 40; i++) {
      await tester.runAsync(
          () => Future<void>.delayed(const Duration(milliseconds: 80)));
      await tester.pump(const Duration(milliseconds: 100));
    }
    await shot('$which-$tag-fight');
  });
}
