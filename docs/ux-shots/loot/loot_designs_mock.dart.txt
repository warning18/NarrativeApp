class _T {
  const _T(this.fr, this.en);
  final String fr, en;
  String of(bool fr_) => fr_ ? fr : en;
}

const _title = _T('Butin', 'Spoils');
const _tier = _T("Coffre d'argent", 'Silver chest');
const _allYours = _T('Tout est à vous.', 'Everything is yours.');
const _take = _T('Prendre', 'Take');
const _gold = _T('+18 or', '+18 gold');
const _knuckle = _T('Osselet du joueur', "Gambler's Knucklebone");
const _knuckleText = _T(
    "Consumé au début d'un combat : une relance supplémentaire à chaque tour de ce combat.",
    'Burned at the start of a fight: one extra reroll each round of that fight.');
const _antidote = _T('Antidote', 'Antidote');
const _antidoteText = _T(
    'Se boit pour guérir le poison et les afflictions.',
    'Drunk to cure poison and afflictions.');
const _goldText = _T('Ajouté à votre bourse.', 'Added to your purse.');
const _useNow = _T('Utiliser tout de suite', 'Use right away');
const _chosen = _T('Choisir', 'Choose');

class _Item {
  const _Item(this.name, this.text, this.icon, this.tone);
  final _T name, text;
  final Widget icon;
  final Color tone;
}

Widget _gold18(double s) => Container(
      width: s * .8,
      height: s * .8,
      decoration: const BoxDecoration(
          color: Color(0xFFF2A93B), shape: BoxShape.circle),
      alignment: Alignment.center,
      child: Text('\$',
          style: TextStyle(
              fontSize: s * .5,
              fontWeight: FontWeight.w800,
              color: const Color(0xFF3A2A10))),
    );

List<_Item> _items(double s) => [
      _Item(_gold, _goldText, _gold18(s), const Color(0xFFF2A93B)),
      _Item(_knuckle, _knuckleText,
          ItemPixelIcon('charm_fourth_roll', 'Charm', size: s), const Color(0xFFE8DCC0)),
      _Item(_antidote, _antidoteText,
          ItemPixelIcon('antidote', 'Potion', size: s), const Color(0xFF7DBE6A)),
    ];

TextStyle _prose({double size = 13, Color? color}) =>
    TextStyle(fontFamily: InkFonts.prose, fontSize: size, height: 1.3, color: color ?? Colors.white70);
TextStyle _sys(double size, {Color? color, FontWeight? w}) =>
    TextStyle(fontFamily: InkFonts.system, fontSize: size, color: color ?? Colors.white, fontWeight: w);
TextStyle _disp(double size) =>
    TextStyle(fontFamily: InkFonts.display, fontSize: size, color: Colors.white);

Widget _modal(Widget child, {double? height}) => Container(
      margin: const EdgeInsets.symmetric(horizontal: 12, vertical: 28),
      padding: const EdgeInsets.all(16),
      decoration: BoxDecoration(
        color: const Color(0xFF1B1923),
        borderRadius: BorderRadius.circular(6),
        border: Border.all(color: const Color(0xFF3B3743)),
      ),
      child: child,
    );

Widget _tierChip(bool fr) => Container(
      padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 5),
      decoration: BoxDecoration(
          borderRadius: BorderRadius.circular(8),
          border: Border.all(color: const Color(0xFFA8A194))),
      child: Text(_tier.of(fr), style: _sys(12, color: const Color(0xFFD7D3CA))),
    );

Widget _takeBtn(bool fr, {String? label, bool wide = false}) => SizedBox(
      width: wide ? double.infinity : null,
      child: FilledButton(
        onPressed: () {},
        style: FilledButton.styleFrom(
            backgroundColor: const Color(0xFFF2C14E),
            foregroundColor: const Color(0xFF1B1923),
            padding: const EdgeInsets.symmetric(horizontal: 28, vertical: 14),
            shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(4))),
        child: Text(label ?? _take.of(fr), style: _sys(15, color: const Color(0xFF1B1923), w: FontWeight.w600)),
      ),
    );

// ---- A: the chest opens and the rewards come out as cards with their text ----
Widget designA(bool fr, {bool pick = false}) {
  final items = _items(40);
  return _modal(Column(mainAxisSize: MainAxisSize.min, children: [
    Row(children: [
      Text(pick ? (fr ? 'Récompense' : 'Reward') : _title.of(fr), style: _disp(26)),
      const Spacer(),
      if (!pick) _tierChip(fr) else Text(fr ? 'Étage 7' : 'Floor 7', style: _sys(13, color: const Color(0xFFD7D3CA))),
    ]),
    const SizedBox(height: 6),
    Stack(alignment: Alignment.center, children: [
      Container(
        width: 150,
        height: 70,
        decoration: BoxDecoration(
            gradient: RadialGradient(colors: [const Color(0x66F2C14E), const Color(0x00F2C14E)])),
      ),
      const PixelIcon('assets/icons/chests/silver_open.png', size: 72),
    ]),
    Text(pick ? (fr ? 'Choisissez une récompense.' : 'Choose one reward.') : _allYours.of(fr),
        style: _prose(size: 14, color: Colors.white)),
    const SizedBox(height: 10),
    for (var i = 0; i < items.length; i++)
      Padding(
        padding: const EdgeInsets.only(bottom: 8),
        child: _cardA(items[i], fr, pick: pick, dim: pick && i != 1, first: i == 0),
      ),
    const SizedBox(height: 6),
    if (!pick) Align(alignment: Alignment.centerRight, child: _takeBtn(fr)),
  ]));
}

Widget _cardA(_Item it, bool fr, {required bool pick, required bool dim, required bool first}) {
  final ring = pick && !dim;
  return Opacity(
    opacity: dim ? .55 : 1,
    child: Container(
      padding: const EdgeInsets.all(10),
      decoration: BoxDecoration(
        color: const Color(0xFF24222E),
        borderRadius: BorderRadius.circular(10),
        border: Border.all(color: ring ? const Color(0xFFF2C14E) : it.tone.withValues(alpha: .55), width: ring ? 2 : 1),
      ),
      child: Row(crossAxisAlignment: CrossAxisAlignment.start, children: [
        SizedBox(width: 44, height: 44, child: Center(child: it.icon)),
        const SizedBox(width: 10),
        Expanded(
          child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
            Text(it.name.of(fr), style: _disp(17)),
            const SizedBox(height: 2),
            Text(it.text.of(fr), style: _prose(size: 12.5)),
            if (!pick && !first) ...[
              const SizedBox(height: 6),
              Row(children: [
                Icon(Icons.bolt, size: 16, color: it.tone),
                const SizedBox(width: 4),
                Text(_useNow.of(fr), style: _sys(12, color: it.tone)),
              ]),
            ],
            if (pick) ...[
              const SizedBox(height: 8),
              Align(
                  alignment: Alignment.centerRight,
                  child: OutlinedButton(
                      onPressed: () {},
                      style: OutlinedButton.styleFrom(
                          side: BorderSide(color: ring ? const Color(0xFFF2C14E) : Colors.white24),
                          foregroundColor: ring ? const Color(0xFFF2C14E) : Colors.white54),
                      child: Text(_chosen.of(fr), style: _sys(13)))),
            ],
          ]),
        ),
      ]),
    ),
  );
}

// ---- B: a compact strip of rewards, a detail line for the selected one ----
Widget designB(bool fr) {
  final items = _items(34);
  const sel = 1;
  return _modal(Column(mainAxisSize: MainAxisSize.min, crossAxisAlignment: CrossAxisAlignment.start, children: [
    Row(children: [
      const PixelIcon('assets/icons/chests/silver_open.png', size: 44),
      const SizedBox(width: 10),
      Text(_title.of(fr), style: _disp(26)),
      const Spacer(),
      _tierChip(fr),
    ]),
    const SizedBox(height: 14),
    IntrinsicHeight(child: Row(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
      for (var i = 0; i < items.length; i++)
        Expanded(
          child: Padding(
            padding: EdgeInsets.only(right: i < 2 ? 8 : 0),
            child: Container(
              padding: const EdgeInsets.symmetric(vertical: 10, horizontal: 4),
              decoration: BoxDecoration(
                color: const Color(0xFF24222E),
                borderRadius: BorderRadius.circular(10),
                border: Border.all(color: i == sel ? const Color(0xFFF2C14E) : Colors.white24, width: i == sel ? 2 : 1),
              ),
              child: Column(children: [
                items[i].icon,
                const SizedBox(height: 6),
                Text(items[i].name.of(fr), textAlign: TextAlign.center, maxLines: 2, style: _sys(12)),
              ]),
            ),
          ),
        ),
    ])),
    const SizedBox(height: 12),
    Container(
      width: double.infinity,
      padding: const EdgeInsets.all(12),
      decoration: BoxDecoration(
          color: const Color(0xFF14131A), borderRadius: BorderRadius.circular(8), border: Border.all(color: Colors.white12)),
      child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
        Text(items[sel].name.of(fr), style: _disp(18)),
        const SizedBox(height: 4),
        Text(items[sel].text.of(fr), style: _prose(size: 13)),
        const SizedBox(height: 8),
        Row(children: [
          Icon(Icons.bolt, size: 16, color: items[sel].tone),
          const SizedBox(width: 4),
          Text(_useNow.of(fr), style: _sys(12, color: items[sel].tone)),
          const Spacer(),
          Switch(value: true, onChanged: (_) {}),
        ]),
      ]),
    ),
    const SizedBox(height: 14),
    Align(alignment: Alignment.centerRight, child: _takeBtn(fr)),
  ]));
}

Future<void> _shot(WidgetTester tester, Widget app, double w, String name) async {
  final key = GlobalKey();
  tester.view.devicePixelRatio = 2;
  tester.view.physicalSize = Size(w * 2, 800 * 2);
  await tester.pumpWidget(MaterialApp(
    debugShowCheckedModeBanner: false,
    theme: buildAppTheme(AppPalette.stitchedInk.schemeFor(Brightness.dark)),
    home: RepaintBoundary(
      key: key,
      child: Scaffold(
        backgroundColor: const Color(0xFF14131A),
        body: Stack(children: [
          // The fight behind, dimmed.
          Positioned.fill(child: Container(color: const Color(0xFF2A2833))),
          Align(alignment: Alignment.topCenter, child: Container(height: 60, color: const Color(0xFF14131A), alignment: Alignment.center, child: Text('Combat : Élite Bandit des rues', style: _disp(15).copyWith(color: Colors.white38)))),
          Positioned.fill(child: Container(color: Colors.black54)),
          Align(alignment: Alignment.center, child: name.startsWith('C') ? SizedBox(height: 800, child: app) : SingleChildScrollView(child: app)),
        ]),
      ),
    ),
  ));
  await tester.runAsync(() => Future<void>.delayed(const Duration(milliseconds: 300)));
  await tester.pump(const Duration(milliseconds: 100));
  await tester.runAsync(() async {
    final b = key.currentContext!.findRenderObject()! as RenderRepaintBoundary;
    final image = await b.toImage(pixelRatio: 1);
    final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
    File('${Platform.environment['OUT']}/$name.png').writeAsBytesSync(bytes!.buffer.asUint8List());
  });
}

void main() {
  testWidgets('loot designs', (tester) async {
    await tester.runAsync(loadAppFonts);
    Directory(Platform.environment['OUT']!).createSync(recursive: true);
    final sets = <String, Widget Function(bool)>{
      'A': (fr) => designA(fr),
      'A-pick': (fr) => designA(fr, pick: true),
      'B': designB,
      'C': designC,
    };
    for (final e in sets.entries) {
      for (final w in [360.0, 412.0]) {
        for (final fr in [true, false]) {
          await _shot(tester, e.value(fr), w, '${e.key}-${w.round()}-${fr ? 'fr' : 'en'}');
        }
      }
    }
  });
}
