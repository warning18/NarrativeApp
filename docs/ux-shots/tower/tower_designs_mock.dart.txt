/// One direction: a name, colours and frames. A is today's look (ink panels,
/// gold, rounded cards); B is bolder (stone, ember, a vertical ladder).
class Dir {
  const Dir(this.name, this.bg, this.panel, this.edge, this.accent, this.onAccent,
      {required this.radius, required this.bold});
  final String name;
  final List<Color> bg;
  final Color panel, edge, accent, onAccent;
  final double radius;
  final bool bold;
}

TextStyle disp(double s, [Color c = Colors.white]) =>

Widget tower(double s, Color c) => CustomPaint(size: Size(s, s * 1.3), painter: _TowerPainter(c));

class _TowerPainter extends CustomPainter {
  _TowerPainter(this.c);
  final Color c;
  @override
  void paint(Canvas canvas, Size s) {
    final p = Paint()..color = c;
    final u = s.width / 12;
    // A stepped pixel tower with a lit window and a crown of teeth.
    canvas.drawRect(Rect.fromLTWH(3 * u, 4 * u, 6 * u, 12 * u), p);
    canvas.drawRect(Rect.fromLTWH(2 * u, 2 * u, 8 * u, 2 * u), p);
    for (var i = 0; i < 4; i++) {
      canvas.drawRect(Rect.fromLTWH((2 + i * 2.2) * u, 0.6 * u, 1.4 * u, 1.6 * u), p);
    }
    canvas.drawRect(Rect.fromLTWH(1 * u, 14 * u, 10 * u, 2 * u), p);
    final w = Paint()..color = const Color(0xFF14131A);
    canvas.drawRect(Rect.fromLTWH(5.2 * u, 6 * u, 1.6 * u, 3 * u), w);
    canvas.drawRect(Rect.fromLTWH(5.2 * u, 11 * u, 1.6 * u, 3 * u), w);
  }
  @override
  bool shouldRepaint(_TowerPainter o) => o.c != c;
}

// ---- 3 & 5. A floor and a boss floor ----
Widget ladder(Dir d, int floor, {bool vertical = false}) {
  if (!vertical) {
    return Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
      Row(mainAxisAlignment: MainAxisAlignment.spaceBetween, children: [
        Text('${floor <= 30 ? floor : 30} / 30', style: sys(13, c: d.accent, w: FontWeight.w600)),
        Text(t(true, '', ''), style: sys(12)),
      ]),
      const SizedBox(height: 4),
      SizedBox(
        height: 18,
        child: Row(children: [
          for (var i = 1; i <= 30; i++)
            Expanded(
              child: Container(
                margin: const EdgeInsets.symmetric(horizontal: .5),
                height: i % 10 == 0 ? 18 : 10,
                alignment: Alignment.bottomCenter,
                child: Container(height: i % 10 == 0 ? 18 : 10, color: i < floor ? d.accent : (i == floor ? Colors.white : Colors.white24)),
              ),
            ),
        ]),
      ),
    ]);
  }
  return SizedBox(
    width: 42,
    child: Column(children: [
      for (var i = 30; i >= 1; i--)
        Expanded(
          child: Container(
            margin: const EdgeInsets.symmetric(vertical: .5),
            alignment: Alignment.center,
            width: i % 10 == 0 ? 40 : 22,
            color: i < floor ? d.accent.withValues(alpha: .7) : (i == floor ? Colors.white : Colors.white12),
            child: i % 10 == 0 || i == floor ? Text('$i', style: sys(10.5, c: i == floor ? Colors.black : Colors.white, w: FontWeight.w700)) : null,
          ),
        ),
    ]),
  );
}

Future<void> _shot(WidgetTester tester, Widget app, double w, String path) async {
  final key = GlobalKey();
  tester.view.devicePixelRatio = 2;
  tester.view.physicalSize = Size(w * 2, 780 * 2);
  await tester.pumpWidget(MaterialApp(
    debugShowCheckedModeBanner: false,
    theme: buildAppTheme(AppPalette.stitchedInk.schemeFor(Brightness.dark)),
    home: RepaintBoundary(key: key, child: Scaffold(body: app)),
  ));
  await tester.runAsync(() => Future<void>.delayed(const Duration(milliseconds: 300)));
  await tester.pump(const Duration(milliseconds: 100));
  await tester.runAsync(() async {
    final b = key.currentContext!.findRenderObject()! as RenderRepaintBoundary;
    final image = await b.toImage(pixelRatio: 1);
    final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
    File(path).writeAsBytesSync(bytes!.buffer.asUint8List());
  });
}

void main() {
  testWidgets('tower designs', (tester) async {
    await tester.runAsync(loadAppFonts);
    final out = Platform.environment['OUT']!;
    Directory(out).createSync(recursive: true);
    final screens = <String, Widget Function(Dir, bool)>{
      '1-menu': menu,
      '2-start': start,
      '3-floor': (d, fr) => floorScreen(d, fr, floor: 7),
      '4-reward': reward,
      '5-boss': (d, fr) => floorScreen(d, fr, floor: 10, boss: true),
      '6a-death': (d, fr) => endScreen(d, fr, win: false),
      '6b-victory': (d, fr) => endScreen(d, fr, win: true),
      '7-marks': marks,
    };
    for (final e in screens.entries) {
      for (final d in [dirA, dirB]) {
        // French at 360 px and English at 412 px: the two ends of the phones
        // (and the longest text against the narrowest screen).
        await _shot(tester, e.value(d, true), 360, '$out/${e.key}-${d.name}-fr-360.png');
        await _shot(tester, e.value(d, false), 412, '$out/${e.key}-${d.name}-en-412.png');
      }
    }
  });
}
