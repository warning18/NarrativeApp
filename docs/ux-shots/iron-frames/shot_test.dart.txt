// the pop-in of an effect icon.

final _k = GlobalKey();
bool _dark = false;
Future<void> _enter(WidgetTester tester, {List<SideLayer>? land, double w = 412, String id = 'rat_matriarch', List<String> more = const []}) async {
  SharedPreferences.setMockInitialValues({
    // The dice thrown at 4x: the tests wait a quarter of the throw (the
    // default speed is proven by dice_speed_test).
    'combat_dice_speed': 'quad',
    if (_dark) 'app_theme_mode': 'dark',
    'gamedb_dice': jsonEncode({
      ...gamedata('dice.json'),
      'board_die': {
        'diceName': 'board_die',
        'numberOfFaces': 1,
        'faces': [
          {
            'faceName': 'Strike',
            'type': 'Attack',
            'value': 4,
            'element': 'None'
          },
        ],
      },
    }),
  });
  usePhone(tester, Size(w, 860));
  await tester.pumpWidget(ProviderScope(
      overrides: [fightRandomProvider.overrideWithValue(() => Random(7))],
      child: RepaintBoundary(key: _k, child: const MyApp())));
  await settle(tester);
  final container =
      ProviderScope.containerOf(tester.element(find.byType(MaterialApp)));
  await tester.runAsync(() => container
      .read(playerSessionProvider.notifier)
      .loadSession(PlayerSession.fromJson({
        'raceId': 'human',
        'professionId': 'warrior',
        'ownedDiceIds': ['board_die'],
        'equippedDiceId': 'board_die',
        'maxHealth': 400,
        'currentHealth': 400,
      })));
  final thug = {
    ...gamedata('enemies.json')[id] as Map<String, dynamic>,
    'maxHealth': 900,
    'skillMoves': <dynamic>[],
  };
  tester
      .state<NavigatorState>(find.byType(Navigator).first)
      .push(MaterialPageRoute<bool>(
          builder: (_) => FightScreen(
                enemyId: id,
                landLayers: land,
                enemy: thug,
                additionalEnemyIds: more,
                additionalEnemies: {for (final m in more) m: {...gamedata('enemies.json')[m] as Map<String, dynamic>, 'maxHealth': 900, 'skillMoves': <dynamic>[]}},
                modifiers: const EncounterModifiers(
                    forcedCondition: BattlefieldCondition.highGround,
                    forcedGoal: FightGoal.slay),
              )));
  await settle(tester);
  Future<void> until(bool Function() f, [int n = 150]) async {
    for (var i = 0; i < n && !f(); i++) {
      await tester.runAsync(() => Future<void>.delayed(const Duration(milliseconds: 60)));
      await tester.pump(const Duration(milliseconds: 100));
    }
  }
  await until(() => shows('Enter Battle') || shows('Roll Dice'));
  if (shows('Enter Battle')) {
    await until(() => find.text('Enter Battle').hitTestable().evaluate().isNotEmpty, 40);
    await tester.tap(find.text('Enter Battle'), warnIfMissed: false);
  }
  await until(() => shows('Roll Dice'));
  for (var i = 0; i < 12; i++) {
    await tester.runAsync(() => Future<void>.delayed(const Duration(milliseconds: 80)));
    await tester.pump(const Duration(milliseconds: 100));
  }
}



void main() {
  testWidgets('frame shot', (tester) async {
    final out = Platform.environment['UX_SHOTS'];
    final c = Platform.environment['UX_CASE'];
    if (out == null || c == null) return;
    // material:env:dark|light:width:one|pack
    final p = c.split(':');
    final off = p[0] == 'off';
    final m = off ? FrameMaterial.iron : FrameMaterial.values.byName(p[0]);
    final e = EnvDecor.values.byName(p[1]);
    _dark = p[2] == 'dark';
    final w = double.parse(p[3]);
    final pack = p[4] == 'pack';
    await loadAppFonts();
    if (m != FrameMaterial.values.first || true) {
      FrameTheme.style = off ? null : FrameStyle(material: m, env: e, compactScale: double.parse(Platform.environment['UX_SCALE'] ?? '1'));
    }
    await tester.runAsync(() => FrameImages.preload(m, e));
    await _enter(tester, w: w, more: pack ? ['harbor_rat', 'plague_hound'] : const []);
    final bytes = await tester.runAsync(() async {
      final o = _k.currentContext!.findRenderObject()! as RenderRepaintBoundary;
      final im = await o.toImage(pixelRatio: 2);
      return im.toByteData(format: ui.ImageByteFormat.png);
    });
    File('$out/fight-${p.join('-')}.png').writeAsBytesSync(bytes!.buffer.asUint8List());
  });
}
