import 'dart:io';
import 'dart:ui' as ui;
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'package:narrative_data_app/providers/palette_provider.dart';
import 'package:narrative_data_app/screens/origin_stories_screen.dart';
import 'package:narrative_data_app/screens/race_profession_screen.dart';
import 'package:narrative_data_app/theme/stitched_ink.dart';
import 'support/fonts.dart';
import 'package:narrative_data_app/theme/frame_style.dart';

void main() {
  testWidgets('life shots', (tester) async {
    final out = Platform.environment['UX_SHOTS'];
    if (out == null) return;
    await loadAppFonts();
    final mat = Platform.environment['UX_MAT'];
    if (mat != null) {
      FrameTheme.style = FrameStyle(material: FrameMaterial.values.byName(mat), env: EnvDecor.none, compactScale: 2);
      await tester.runAsync(() => FrameImages.preload(FrameMaterial.values.byName(mat), EnvDecor.none));
    }
    final tag = Platform.environment['UX_TAG'] ?? 'now';
    SharedPreferences.setMockInitialValues({});
    final key = GlobalKey();
    for (final dark in [true, false]) {
      final theme = buildAppTheme(AppPalette.stitchedInk.schemeFor(dark ? Brightness.dark : Brightness.light));
      final tn = dark ? 'dark' : 'light';
      for (final w in [412.0, 360.0]) {
        tester.view.physicalSize = Size(w * 2, 880 * 2); tester.view.devicePixelRatio = 2;
        Future<void> shot(String name) async {
          for (var i = 0; i < 6; i++) { await tester.runAsync(() => Future<void>.delayed(const Duration(milliseconds: 120))); await tester.pump(const Duration(milliseconds: 100)); }
          final bytes = await tester.runAsync(() async {
            final o = key.currentContext!.findRenderObject()! as RenderRepaintBoundary;
            final im = await o.toImage(pixelRatio: 2);
            return im.toByteData(format: ui.ImageByteFormat.png);
          });
          File('$out/$tag-$name-${w.toInt()}-$tn.png').writeAsBytesSync(bytes!.buffer.asUint8List());
        }
        await tester.pumpWidget(ProviderScope(child: MaterialApp(debugShowCheckedModeBanner: false, theme: theme, home: RepaintBoundary(key: key, child: const RaceProfessionScreen()))));
        await shot('race-1');
        final dwarf = find.byKey(const Key('choice_card_dwarf'));
        if (dwarf.evaluate().isNotEmpty) {
          await tester.tap(dwarf); await shot('race-2');
          final next = find.byKey(const Key('rp_next'));
          if (next.evaluate().isNotEmpty) { await tester.tap(next); await shot('race-3'); }
          final warrior = find.byKey(const Key('choice_card_warrior'));
          if (warrior.evaluate().isNotEmpty) { await tester.tap(warrior); await shot('race-4'); }
        }
        await tester.pumpWidget(ProviderScope(child: MaterialApp(debugShowCheckedModeBanner: false, theme: theme, home: RepaintBoundary(key: key, child: const OriginStoriesScreen(raceId: 'dwarf', professionId: 'warrior')))));
        await shot('life-1');
      }
    }
  });
}
